cmake_minimum_required(VERSION 3.16)
project(limit_order_book CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release)
endif()
set(CMAKE_CXX_FLAGS_RELEASE "-O2")
add_library(lob src/map_book.cpp src/ladder_book.cpp)
target_include_directories(lob PUBLIC include)
add_executable(test_books tests/test_books.cpp)
target_link_libraries(test_books lob)
add_executable(bench bench/bench.cpp)
target_link_libraries(bench lob)
enable_testing()
add_test(NAME books COMMAND test_books)
